  skip: string | null,
): number {
  let count = 0;
  for (const s of plan.steps) {
    if (s.id === skip || !usesResource(s, resource)) continue;
    const i = iv[s.id];
    if (i !== undefined && i.startMin <= minute && minute < i.endMin) count++;
  }
  return count;
}

/**
 * RQ2.2: a done step keeps `[startMin, endMin)`; a running step lasts
 * `[startMin, max(startMin + minutes + extraMin, now))`.
 */
export function oraclePinnedIntervals(plan: Plan, progress: Progress, now: number): Iv {
  const out: Iv = {};
  for (const step of byId(plan)) {
    const entry = progress[step.id];
    if (entry === undefined) continue;
    if (entry.state === 'done') {
      out[step.id] = { startMin: entry.startMin, endMin: entry.endMin };
    } else {
      out[step.id] = {
        startMin: entry.startMin,
        endMin: Math.max(entry.startMin + step.minutes + entry.extraMin, now),
      };
    }
  }
  return out;
}

const intervalsOfSteps = (steps: readonly { id: string; startMin: number; endMin: number }[]): Iv => {
  const out: Iv = {};
  for (const s of steps) out[s.id] = { startMin: s.startMin, endMin: s.endMin };
  return out;
}

/**
 * R1 on a rebuilt schedule: `capacity:<resource>:<minute>` for every minute >= now where the
 * use exceeds the kitchen; `precedence:<step>:<prereq>` for every pending step that starts
 * before a prerequisite ends; G3: `missing:`, `duplicate:`, `extra:`, `not_integer:<id>`,
 * `length:<id>` (a pending step not lasting its minutes, or a pinned step differing from its
 * RQ2.2 interval), `after_serving:<id>`; and R2 `before_now:<id>` for a pending step starting
 * before now.
 */
export function oracleRebuildViolations(
  plan: Plan,
  kitchen: Kitchen,
  progress: Progress,
  now: number,
  steps: readonly RebuiltStep[],
  servingMin: number,
): string[] {
  const out: string[] = [];
  const iv = intervalsOfSteps(steps);
  const planIds = byId(plan).map((s) => s.id);
  for (const id of planIds) {
    const count = steps.filter((s) => s.id === id).length;
    if (count === 0) out.push(`missing:${id}`);
    if (count > 1) out.push(`duplicate:${id}`);
  }
  for (const id of [...new Set(steps.map((s) => s.id))].filter((x) => !planIds.includes(x)).sort()) {
    out.push(`extra:${id}`);
  }
  const pinned = oraclePinnedIntervals(plan, progress, now);
  for (const step of byId(plan)) {
    const i = iv[step.id];
    if (i === undefined) continue;
    if (!Number.isInteger(i.startMin) || !Number.isInteger(i.endMin)) out.push(`not_integer:${step.id}`);
    const expected = pinned[step.id];
    if (expected !== undefined) {
      if (i.startMin !== expected.startMin || i.endMin !== expected.endMin) out.push(`length:${step.id}`);
    } else {
      if (i.endMin - i.startMin !== step.minutes) out.push(`length:${step.id}`);
      if (i.startMin < now) out.push(`before_now:${step.id}`);
    }
    if (i.endMin > servingMin) out.push(`after_serving:${step.id}`);
  }
  for (const step of byId(plan)) {
    if (!isPending(progress, step.id)) continue;
    const i = iv[step.id];
    if (i === undefined) continue;
    for (const prereq of [...new Set(step.after)].sort()) {
      const p = iv[prereq];
      if (p !== undefined && i.startMin < p.endMin) out.push(`precedence:${step.id}:${prereq}`);
    }
  }
  let first = Infinity;
  let last = -Infinity;
  for (const s of steps) {
    first = Math.min(first, s.startMin);
    last = Math.max(last, s.endMin);
  }
  for (let minute = Math.max(first, now); minute < last; minute++) {
    for (const resource of RESOURCE_NAMES) {
      const count = useAt(plan, iv, resource, minute, null);
      if (count > limitOf(kitchen, resource)) out.push(`capacity:${resource}:${minute}`);
    }
  }
  return out;
}

/**
 * P-RB4: `overfull:<step>:<resource>:<minute>` for every minute where a pending step is active
 * and the use of that resource, the step included, exceeds the kitchen.
 */
export function oraclePendingOverfull(
  plan: Plan,
  kitchen: Kitchen,
  progress: Progress,
  _now: number,
  steps: readonly RebuiltStep[],
): string[] {
  const out: string[] = [];
  const iv = intervalsOfSteps(steps);
  for (const step of byId(plan)) {
    const i = iv[step.id];
    if (!isPending(progress, step.id) || i === undefined) continue;
    for (const resource of RESOURCE_NAMES) {
      if (!usesResource(step, resource)) continue;
      for (let minute = i.startMin; minute < i.endMin; minute++) {
        if (useAt(plan, iv, resource, minute, null) > limitOf(kitchen, resource)) {
          out.push(`overfull:${step.id}:${resource}:${minute}`);
        }
      }
    }
  }
  return out;
}

/**
 * RQ4: the forward-repair intervals. Pinned steps keep their RQ2.2 interval. Pending steps are
 * placed one at a time: among those whose pending prerequisites are all placed, the smallest
 * `(prev start, id)`; it starts at the first minute >= max(prev start, now, end of each
 * prerequisite) from which every resource it uses has a free unit in every minute (each start
 * is tried in turn), counting the pinned steps and the pending steps already placed.
 */
export function oracleForwardRepair(
  plan: Plan,
  kitchen: Kitchen,
  progress: Progress,
  now: number,
  prevStarts: Starts,
): Iv {
  const iv: Iv = oraclePinnedIntervals(plan, progress, now);
  const unplaced = byId(plan).filter((s) => isPending(progress, s.id));
  while (unplaced.length > 0) {
    const ready = unplaced.filter((s) =>
      s.after.every((p) => iv[p] !== undefined || !isPending(progress, p)),
    );
    let pick: PlanStep | undefined;
    for (const s of ready) {
      if (
        pick === undefined ||
        startOf(prevStarts, s) < startOf(prevStarts, pick) ||
        (startOf(prevStarts, s) === startOf(prevStarts, pick) && s.id < pick.id)
      ) {
        pick = s;
      }
    }
    if (pick === undefined) throw new Error('oracleForwardRepair: no ready step (a cycle?)');
    let lower = Math.max(startOf(prevStarts, pick), now);
    for (const p of pick.after) lower = Math.max(lower, intervalOf(iv, p).endMin);
    let start = lower;
    for (;;) {
      let fits = true;
      for (let minute = start; minute < start + pick.minutes && fits; minute++) {
        for (const resource of RESOURCE_NAMES) {
          if (!usesResource(pick, resource)) continue;
          if (useAt(plan, iv, resource, minute, null) + 1 > limitOf(kitchen, resource)) fits = false;
        }
      }
      if (fits) break;
      start++;
      if (start > 100_000) throw new Error('oracleForwardRepair: no room found');
    }
    iv[pick.id] = { startMin: start, endMin: start + pick.minutes };
    unplaced.splice(unplaced.indexOf(pick), 1);
  }
  return iv;
}

/**
 * RQ8.3: null where the result keeps; a terminal step `servingMin - end`; otherwise
 * `max(0, latest successor start - end)`.
 */
export function oracleRebuildWaits(
  plan: Plan,
  iv: Iv,
  servingMin: number,
): Record<string, number | null> {
  const out: Record<string, number | null> = {};
  for (const s of byId(plan)) {
    if (s.holdMin === null) {
      out[s.id] = null;
      continue;
    }
    const users = usersOf(plan, s.id);
    const end = intervalOf(iv, s.id).endMin;
    if (users.length === 0) {
      out[s.id] = servingMin - end;
      continue;
    }
    let latest = -Infinity;
    for (const u of users) latest = Math.max(latest, intervalOf(iv, u.id).startMin);
    out[s.id] = Math.max(0, latest - end);
  }
  return out;
}

/** RQ8.4: every step whose wait is longer than its hold, by id. */
export function oracleRebuildHoldExceeded(
  plan: Plan,
  iv: Iv,
  servingMin: number,
): { stepId: string; waitMin: number; holdMin: number }[] {
  const waits = oracleRebuildWaits(plan, iv, servingMin);
  const out: { stepId: string; waitMin: number; holdMin: number }[] = [];
  for (const s of byId(plan)) {
    const wait = waits[s.id];
    if (wait === null || wait === undefined || s.holdMin === null) continue;
    if (wait > s.holdMin) out.push({ stepId: s.id, waitMin: wait, holdMin: s.holdMin });
  }
  return out;
}

/**
 * RQ8.1 and RQ8.2: 0 for a pinned step. For a pending step, the largest k >= 0 such that
 * delaying it alone by each d in 1..k keeps it ending at or before `servingMin`, before the
 * start of each pending successor, and with a free unit of every resource it uses in every
 * minute of its new window (all other intervals counted). Tries d = 1, 2, ... until one fails.
 */
export function oracleRebuildSlack(
  plan: Plan,
  kitchen: Kitchen,
  progress: Progress,
  _now: number,
  iv: Iv,
  servingMin: number,
  stepId: string,
): number {
  const step = plan.steps.find((s) => s.id === stepId);
  if (step === undefined || !isPending(progress, stepId)) return 0;
  const here = intervalOf(iv, stepId);
  const pendingUsers = usersOf(plan, stepId).filter((u) => isPending(progress, u.id));
  for (let d = 1; ; d++) {
    const newStart = here.startMin + d;
    const newEnd = here.endMin + d;
    if (newEnd > servingMin) return d - 1;
    if (pendingUsers.some((u) => intervalOf(iv, u.id).startMin < newEnd)) return d - 1;
    for (let minute = newStart; minute < newEnd; minute++) {
      for (const resource of RESOURCE_NAMES) {
        if (!usesResource(step, resource)) continue;
        if (useAt(plan, iv, resource, minute, stepId) + 1 > limitOf(kitchen, resource)) return d - 1;
      }
    }
  }
}

/**
 * RQ7: the field and range violations of `oracleLaneViolations`, and the lane clashes that
 * RQ7.2 does not allow. A clash of two overlapping steps on one lane is allowed only when the
 * one that comes later in `(start, id)` order is pinned and the use of that resource at its
 * start exceeds the kitchen.
 */
export function oracleRebuildLaneViolations(
  plan: Plan,
  kitchen: Kitchen,
  iv: Iv,
  pinned: readonly string[],
  steps: readonly RebuiltStep[],
): string[] {
  return oracleLaneViolations(plan, kitchen, steps).filter((v) => {
    const parts = v.split(':');
    if (parts[0] !== 'lane_clash') return true;
    const resource = parts[1] as ResourceName;
    const a = parts[3] ?? '';
    const b = parts[4] ?? '';
    const ia = intervalOf(iv, a);
    const ib = intervalOf(iv, b);
    const bIsLater = ib.startMin > ia.startMin || (ib.startMin === ia.startMin && b > a);
    const later = bIsLater ? b : a;
    const laterStart = bIsLater ? ib.startMin : ia.startMin;
    const allowed =
      pinned.includes(later) &&
      useAt(plan, iv, resource, laterStart, null) > limitOf(kitchen, resource);
    return !allowed;
  });
}

/**
 * Plays the meal: base facts are placed first, then the other steps one at a time, the ready
 * step with the smallest `(planned start + shift, id)` first, each at the first minute from
 * which its resources are free for its real duration (every minute counts as free with
 * `overCapacity`). Returns the real interval of every step.
 */
function playRun(
  s: AnySchedule,
  draws: readonly RunDraw[],
  opts: RunOpts,
): Map<string, Actual> {
  const steps = [...s.plan.steps].sort((a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
  const drawOf = new Map<string, RunDraw>(steps.map((x, i) => [x.id, draws[i] as RunDraw]));
  const kitchenOf: Record<'burner' | 'oven' | 'cook', number> = {
    burner: s.kitchen.burners,
    oven: s.kitchen.ovens,
    cook: s.kitchen.cooks,
  };
  const used = new Map<string, number>();
  const placed = new Map<string, Actual>();
  const resourcesOfStep = (step: Step): ('burner' | 'oven' | 'cook')[] => {
    const out: ('burner' | 'oven' | 'cook')[] = [];
    if (step.appliance !== 'none') out.push(step.appliance);
    if (step.handsOn) out.push('cook');
    return out;
  };
  const place = (step: Step, startMin: number, endMin: number): void => {
    placed.set(step.id, { startMin, endMin });
    for (const r of resourcesOfStep(step)) {
      for (let minute = startMin; minute < endMin; minute++) {
        used.set(`${r}:${minute}`, (used.get(`${r}:${minute}`) ?? 0) + 1);
      }
    }
  };
  const plannedStart = (id: string): number => s.steps.find((x) => x.id === id)?.startMin ?? 0;
  const floorMin = opts.base === undefined ? AXIS_MIN : opts.base.now;

  // Facts of an earlier rebuild stay.
  for (const step of steps) {
    const fact = opts.base?.progress[step.id];
    const draw = drawOf.get(step.id);
    if (fact === undefined || draw === undefined || opts.base === undefined) continue;
    if (fact.state === 'done') {
      place(step, fact.startMin, fact.endMin);
    } else {
      const expected = Math.max(fact.startMin + step.minutes + fact.extraMin, opts.base.now);
      place(step, fact.startMin, Math.max(opts.base.now, expected + draw.endShift));
    }
  }

  const todo = steps.filter((x) => !placed.has(x.id));
  while (todo.length > 0) {
    let pick: Step | undefined;
    let pickDesired = 0;
    for (const step of todo) {
      const draw = drawOf.get(step.id) as RunDraw;
      if (!draw.ignoresAfter && !step.after.every((p) => placed.has(p))) continue;
      const desired = Math.max(floorMin, plannedStart(step.id) + draw.startShift);
      if (pick === undefined || desired < pickDesired || (desired === pickDesired && step.id < pick.id)) {
        pick = step;
        pickDesired = desired;
      }
    }
    if (pick === undefined) throw new Error('playRun: no ready step');
    const draw = drawOf.get(pick.id) as RunDraw;
    const duration = Math.max(0, pick.minutes + draw.durDelta);
    let start = pickDesired;
    if (!draw.ignoresAfter) {
      for (const p of pick.after) start = Math.max(start, placed.get(p)?.endMin ?? start);
    }
    for (;;) {
      if (opts.overCapacity === true || duration === 0) break;
      let full: number | null = null;
      for (let minute = start; minute < start + duration && full === null; minute++) {
        for (const r of resourcesOfStep(pick)) {
          if ((used.get(`${r}:${minute}`) ?? 0) + 1 > kitchenOf[r]) full = minute;
        }
      }
      if (full === null) break;
      start = full + 1;
    }
    place(pick, start, start + duration);
    todo.splice(todo.indexOf(pick), 1);
  }
  return placed;
}

/**
 * The cook's run as progress at `now`: a step that has not started is absent, one that ended
 * is done at its real times, one under way is running with the minutes the cook tapped.
 */
function cutRun(
  s: AnySchedule,
  actual: ReadonlyMap<string, Actual>,
  draws: readonly RunDraw[],
  now: number,
  base: Run | undefined,
): Progress {
  const steps = [...s.plan.steps].sort((a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
  const progress: Progress = {};
  steps.forEach((step, i) => {
    const real = actual.get(step.id);
    const draw = draws[i] as RunDraw;
    if (real === undefined || real.startMin > now) return;
    if (real.endMin <= now) {
      progress[step.id] = { state: 'done', startMin: real.startMin, endMin: real.endMin };
      return;
    }
    const before = base?.progress[step.id];
    const alreadyTapped = before?.state === 'running' ? before.extraMin : 0;
    progress[step.id] = {
      state: 'running',
      startMin: real.startMin,
      extraMin: Math.min(1440, alreadyTapped + draw.taps),
    };
  });
  return progress;
}

/**
 * Plays a meal on `s` and cuts it at a `now`: early and late starts, steps that take longer
 * or shorter, running steps with or without tapped minutes. A run respects precedence and
 * capacity unless `outOfOrder` or `overCapacity` say otherwise; with `base` the facts of an
 * earlier rebuild stay and the meal goes on from its `now`.
 */
export function arbProgress(s: AnySchedule, opts: RunOpts = {}): fc.Arbitrary<Run> {
  const n = s.plan.steps.length;
  const arbDraws = fc.array(arbRunDraw(opts.outOfOrder === true), { minLength: n, maxLength: n });
  const arbNowDraw =
    opts.base === undefined
      ? fc.tuple(fc.integer({ min: 0, max: 7 }), arbNow(s))
      : fc.tuple(fc.constant(0), fc.integer({ min: 0, max: 90 }));
  return fc.tuple(arbDraws, arbNowDraw).map(([draws, [pickNow, drawn]]): Run => {
    const actual = playRun(s, draws, opts);
    let now: number;
    if (opts.base !== undefined) {
      now = Math.min(REBUILD_MAX_MIN, opts.base.now + drawn);
    } else if (pickNow === 6) {
      now = clampMinute(Math.max(...[...actual.values()].map((x) => x.endMin)));
    } else if (pickNow === 7) {
      now = clampMinute(Math.min(...s.steps.map((x) => x.startMin)) - 1);
    } else {
      now = drawn;
    }
    return { progress: cutRun(s, actual, draws, now, opts.base), now };
  });
}

/**
 * The cook goes on exactly as `s` says up to `now`, on top of what `base` already knows: a
 * pending step planned to start before `now` is done at its planned interval if that ends by
 * `now`, else running with no extra minutes; a running step whose interval in `s` ends by
 * `now` is done at it; everything else is kept.
 */
export function followPlan(s: AnySchedule, base: Progress, now: number): Progress {
  const progress: Progress = { ...base };
  for (const step of s.steps) {
    const known: ProgressEntry | undefined = base[step.id];
    if (known?.state === 'done') continue;
    if (known === undefined && step.startMin >= now) continue;
    if (step.endMin <= now) {
      progress[step.id] = { state: 'done', startMin: step.startMin, endMin: step.endMin };
    } else if (known === undefined) {
      progress[step.id] = { state: 'running', startMin: step.startMin, extraMin: 0 };
    }
  }
  return progress;
}

/** A cook who follows the schedule exactly up to some `now` (the premise of R4). */
export function arbOnPlan(s: Schedule): fc.Arbitrary<Run> {
  return arbNow(s).map((now): Run => ({ progress: followPlan(s, {}, now), now }));
}

 * the earliest start at or after `max(prev start, now, end of each prerequisite)` where its
 * resources are free, counting the pinned steps from `nowMin` on and the pending steps already
 * placed. A pending step is ready when every pending prerequisite is placed; among the ready
 * steps the smallest `(prev start, id)` goes first. No pending step starts before its previous
 * start or before `nowMin`. `plan` must be canonical and valid; `prevStarts` has every step.
 */
export function forwardRepair(
  plan: Plan,
  cap: Capacity,
  prevStarts: ReadonlyMap<string, number>,
  pinned: ReadonlyMap<string, Interval>,
  nowMin: number,
): Map<string, Interval> {
  const intervals = new Map<string, Interval>(pinned);
  const prevOf = (step: Step): number => {
    const value = prevStarts.get(step.id);
    if (value === undefined) throw new Error(`forwardRepair: no previous start for ${step.id}`);
    return value;
  };
  const tl = timelineFrom(plan, cap, pinned, nowMin);
  const unplaced = plan.steps.filter((s) => !pinned.has(s.id));
  const total = plan.steps.reduce((sum, s) => sum + s.minutes, 0);
  let horizon = nowMin;
  for (const i of pinned.values()) horizon = Math.max(horizon, i.endMin);
  for (const s of unplaced) horizon = Math.max(horizon, prevOf(s));
  const ceilMin = horizon + total;

  while (unplaced.length > 0) {
    let pick: Step | undefined;
    for (const s of unplaced) {
      if (!s.after.every((p) => intervals.has(p))) continue;
      if (
        pick === undefined ||
        prevOf(s) < prevOf(pick) ||
        (prevOf(s) === prevOf(pick) && compareIds(s.id, pick.id) < 0)
      ) {
        pick = s;
      }
    }
    if (pick === undefined) throw new Error('forwardRepair: no ready step (a cycle?)');
    let lower = Math.max(prevOf(pick), nowMin);
    for (const p of pick.after) lower = Math.max(lower, intervals.get(p)?.endMin ?? lower);

    const startMin = earliestFit(tl, rs, pick.minutes, lower, ceilMin);
    intervals.set(pick.id, { startMin, endMin: startMin + pick.minutes });
    tl.add(rs, startMin, startMin + pick.minutes);
    unplaced.splice(unplaced.indexOf(pick), 1);

}

 * Rebuild lanes. Steps go in order of (start, id). For each resource a step uses, the lane is
 * the first of these that no overlapping, already assigned step holds: its lane in the previous
 * schedule (when within the kitchen), the lane of a prerequisite that uses the resource (latest
 * end first, ties by id), the smallest lane. When every lane is held, which only a pinned step
 * can meet, it keeps its previous lane if within the kitchen, else lane 1; this is not an error.
 * `plan` must be canonical and valid and `intervals` must hold every step.
 */
export function assignRebuildLanes(
  plan: Plan,
  cap: Capacity,
  intervals: ReadonlyMap<string, Interval>,
  prevLanes: ReadonlyMap<string, StepLanes>,
): Map<string, StepLanes> {
  const byId = new Map<string, Step>(plan.steps.map((s) => [s.id, s]));
  const intervalOf = (id: string): Interval => {
    const value = intervals.get(id);
    if (value === undefined) throw new Error(`assignRebuildLanes: no interval for ${id}`);
    return value;
  };
  const lanes = new Map<string, StepLanes>();
  const laneOf = (id: string, r: Resource): number | null => lanes.get(id)?.[r] ?? null;
  const prevLaneOf = (id: string, r: Resource): number | null => {
    const lane = prevLanes.get(id)?.[r] ?? null;
    return lane !== null && lane >= 1 && lane <= cap[r] ? lane : null;
  };

  const order = [...plan.steps].sort(
    (a, b) =>
      intervalOf(a.id).startMin - intervalOf(b.id).startMin || compareIds(a.id, b.id),
  );
  const assigned: Step[] = [];
  for (const step of order) {
    const here = intervalOf(step.id);
    const chosen: Record<Resource, number | null> = { burner: null, oven: null, cook: null };
    for (const r of resourcesOf(step)) {
      const busy = new Set<number>();
      for (const u of assigned) {
        if (!resourcesOf(u).includes(r)) continue;
        const other = intervalOf(u.id);
        if (other.startMin >= here.endMin || here.startMin >= other.endMin) continue;
        const lane = laneOf(u.id, r);
        if (lane !== null) busy.add(lane);
      }
      const candidates: number[] = [];
      const prev = prevLaneOf(step.id, r);
      if (prev !== null) candidates.push(prev);
      const prereqs = step.after
        .map((p) => byId.get(p))
        .filter((p): p is Step => p !== undefined && resourcesOf(p).includes(r))
        .sort((a, b) => intervalOf(b.id).endMin - intervalOf(a.id).endMin || compareIds(a.id, b.id));
      for (const p of prereqs) {
        const lane = laneOf(p.id, r);
        if (lane !== null) candidates.push(lane);
      }
      let lane = candidates.find((l) => !busy.has(l)) ?? null;
      for (let l = 1; lane === null && l <= cap[r]; l++) {
        if (!busy.has(l)) lane = l;
      }
      chosen[r] = lane ?? prev ?? 1;
    }
    lanes.set(step.id, chosen);
    assigned.push(step);

export function rebuildReadout(
  plan: Plan,
  cap: Capacity,
  intervals: ReadonlyMap<string, Interval>,
  pinned: ReadonlySet<string>,
  nowMin: number,
  servingMin: number,
): Readout {
  const succ = successors(plan);
  const intervalOf = (id: string): Interval => {
    const value = intervals.get(id);
    if (value === undefined) throw new Error(`rebuildReadout: no interval for ${id}`);
    return value;

  const slackMin = new Map<string, number>();
  const waitMin = new Map<string, number | null>();
  const warnings: HoldExceededIssue[] = [];
  for (const step of plan.steps) {
    const here = intervalOf(step.id);
    const users = succ.get(step.id) ?? [];
    const rs = resourcesOf(step);

    let slack = 0;
    if (!pinned.has(step.id)) {
      let deadline = servingMin;
      for (const u of users) {

      slack = Math.max(0, limit);

          slack = d - 1;
          break;
        }
      }
      tl.add(rs, Math.max(here.startMin, nowMin), here.endMin);

        waitMin: wait,
        holdMin: step.holdMin,

## Decisions I made

- **R1 oracle:** `oracleRebuildViolations` checks plain G1 from `now`, as T2.2 requires for E7. R1 as worded therefore applies to capacity-respecting runs only. `oraclePendingOverfull` covers the other case.
- **Empty pinned intervals:** In the lane oracle, an empty pinned interval inside a pending step's window counts as overlapping. That matches the existing lane oracle, and the algorithm treats it the same way.
- **Docs left as they were:** `scheduler/design.md` line 616 and `scheduler/tasks.md` T3.1 still use G9 as the example unknown id. The task allowed only the property table in the design.
