// Scheduler oracles. Written from scheduler-guarantees.md: plain loops over every minute and
// every step, nothing shared with src/core/schedule.ts, passes/ or timeline.ts.

export type Starts = Record<string, number>;
type ResourceName = 'burner' | 'oven' | 'cook';
const RESOURCE_NAMES: readonly ResourceName[] = ['burner', 'oven', 'cook'];

type PlanStep = Plan['steps'][number];

const startOf = (starts: Starts, step: PlanStep): number => starts[step.id] ?? 0;
const endOf = (starts: Starts, step: PlanStep): number => startOf(starts, step) + step.minutes;
const activeAt = (starts: Starts, step: PlanStep, minute: number): boolean =>
  startOf(starts, step) <= minute && minute < endOf(starts, step);

function usesResource(step: PlanStep, resource: ResourceName): boolean {
  if (resource === 'cook') return step.handsOn;
  return step.appliance === resource;
}

function limitOf(kitchen: Kitchen, resource: ResourceName): number {
  if (resource === 'burner') return kitchen.burners;
  if (resource === 'oven') return kitchen.ovens;
  return kitchen.cooks;
}

/** The steps in plain code-unit order of id. */
function byId(plan: Plan): PlanStep[] {
  return [...plan.steps].sort((a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
}

/** The steps that list `id` in `after`. */
function usersOf(plan: Plan, id: string): PlanStep[] {
  return byId(plan).filter((s) => s.after.includes(id));
}

/** G1: every minute and resource where more steps are active than the kitchen has units. */
export function oracleCapacityViolations(
  plan: Plan,
  kitchen: Kitchen,
  starts: Starts,
): { minute: number; resource: ResourceName; count: number; limit: number }[] {
  const out: { minute: number; resource: ResourceName; count: number; limit: number }[] = [];
  if (plan.steps.length === 0) return out;
  let first = Infinity;
  let last = -Infinity;
  for (const s of plan.steps) {
    first = Math.min(first, startOf(starts, s));
    last = Math.max(last, endOf(starts, s));
  }
  for (let minute = first; minute < last; minute++) {
    for (const resource of RESOURCE_NAMES) {
      let count = 0;
      for (const s of plan.steps) {
        if (usesResource(s, resource) && activeAt(starts, s, minute)) count++;
      }
      const limit = limitOf(kitchen, resource);
      if (count > limit) out.push({ minute, resource, count, limit });
    }
  }
  return out;
}

/** G2: every step that starts before one of its prerequisites ends. */
export function oraclePrecedenceViolations(
  plan: Plan,
  starts: Starts,
): { stepId: string; prereq: string }[] {
  const out: { stepId: string; prereq: string }[] = [];
  for (const s of byId(plan)) {
    for (const prereq of [...new Set(s.after)].sort()) {
      const p = plan.steps.find((x) => x.id === prereq);
      if (p === undefined) continue;
      if (startOf(starts, s) < endOf(starts, p)) out.push({ stepId: s.id, prereq });
    }
  }
  return out;
}

/**
 * G3: `missing:<id>`, `duplicate:<id>`, `extra:<id>`, then per scheduled step
 * `not_integer:<id>`, `length:<id>` (end - start differs from minutes), `after_serving:<id>`.
 */
export function oracleCompletenessViolations(
  plan: Plan,
  steps: readonly ScheduledStep[],
): string[] {
  const out: string[] = [];
  const planIds = byId(plan).map((s) => s.id);
  for (const id of planIds) {
    const count = steps.filter((s) => s.id === id).length;
    if (count === 0) out.push(`missing:${id}`);
    if (count > 1) out.push(`duplicate:${id}`);
  }
  const extra = [...new Set(steps.map((s) => s.id))].filter((id) => !planIds.includes(id)).sort();
  for (const id of extra) out.push(`extra:${id}`);
  const entries = [...steps].sort((a, b) => (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
  for (const e of entries) {
    const step = plan.steps.find((s) => s.id === e.id);
    if (step === undefined) continue;
    if (!Number.isInteger(e.startMin) || !Number.isInteger(e.endMin)) out.push(`not_integer:${e.id}`);
    if (e.endMin - e.startMin !== step.minutes) out.push(`length:${e.id}`);
    if (e.endMin > 0) out.push(`after_serving:${e.id}`);
  }
  return out;
}

/** G1 to G3 on bare starts: integers, nothing ends after 0, no capacity or precedence violation. */
export function oracleValid(plan: Plan, kitchen: Kitchen, starts: Starts): boolean {
  for (const s of plan.steps) {
    const start = starts[s.id];
    if (start === undefined || !Number.isInteger(start)) return false;
    if (start + s.minutes > 0) return false;
  }
  return (
    oracleCapacityViolations(plan, kitchen, starts).length === 0 &&
    oraclePrecedenceViolations(plan, starts).length === 0
  );
}

/**
 * Notation of scheduler-guarantees.md: null where the result keeps; for a terminal step
 * `0 - end`; otherwise the latest start of its users minus its end.
 */
export function oracleWaits(plan: Plan, starts: Starts): Record<string, number | null> {
  const out: Record<string, number | null> = {};
  for (const s of byId(plan)) {
    if (s.holdMin === null) {
      out[s.id] = null;
      continue;
    }
    const users = usersOf(plan, s.id);
    if (users.length === 0) {
      out[s.id] = 0 - endOf(starts, s);
      continue;
    }
    let latest = -Infinity;
    for (const u of users) latest = Math.max(latest, startOf(starts, u));
    out[s.id] = latest - endOf(starts, s);
  }
  return out;
}

/** G4.3: every step whose wait is longer than its hold, by id. */
export function oracleHoldExceeded(
  plan: Plan,
  starts: Starts,
): { stepId: string; waitMin: number; holdMin: number }[] {
  const waits = oracleWaits(plan, starts);
  const out: { stepId: string; waitMin: number; holdMin: number }[] = [];
  for (const s of byId(plan)) {
    const wait = waits[s.id];
    if (wait === null || wait === undefined || s.holdMin === null) continue;
    if (wait > s.holdMin) out.push({ stepId: s.id, waitMin: wait, holdMin: s.holdMin });
  }
  return out;
}

/**
 * G5.1 and G6: `lane_missing:<id>:<resource>`, `lane_extra:<id>:<resource>`,
 * `lane_range:<id>:<resource>`, then `lane_clash:<resource>:<lane>:<idA>:<idB>` (idA < idB).
 */
export function oracleLaneViolations(
  plan: Plan,
  kitchen: Kitchen,
  steps: readonly ScheduledStep[],
): string[] {
  const out: string[] = [];
  const entryOf = (id: string): ScheduledStep | undefined => steps.find((e) => e.id === id);
  for (const s of byId(plan)) {
    const entry = entryOf(s.id);
    if (entry === undefined) continue;
    for (const resource of RESOURCE_NAMES) {
      const lane = entry.lanes[resource];
      const needed = usesResource(s, resource);
      if (needed && lane === null) out.push(`lane_missing:${s.id}:${resource}`);
      if (!needed && lane !== null) out.push(`lane_extra:${s.id}:${resource}`);
      if (lane !== null && !(Number.isInteger(lane) && lane >= 1 && lane <= limitOf(kitchen, resource))) {
        out.push(`lane_range:${s.id}:${resource}`);
      }
    }
  }
  const ids = byId(plan).map((s) => s.id);
  for (const resource of RESOURCE_NAMES) {
    for (let i = 0; i < ids.length; i++) {
      for (let j = i + 1; j < ids.length; j++) {
        const a = entryOf(ids[i] ?? '');
        const b = entryOf(ids[j] ?? '');
        if (a === undefined || b === undefined) continue;
        const lane = a.lanes[resource];
        if (lane === null || lane !== b.lanes[resource]) continue;
        if (a.startMin < b.endMin && b.startMin < a.endMin) {
          out.push(`lane_clash:${resource}:${lane}:${a.id}:${b.id}`);
        }
      }
    }
  }
  return out;
}

/**
 * G8, by the definition: the largest k >= 0 such that delaying the step alone by each d in
 * 1..k keeps G1 to G3 true (lanes ignored, holds not considered). Tries d = 1, 2, ... and
 * stops at the first failure.
 */
export function oracleSlack(plan: Plan, kitchen: Kitchen, starts: Starts, stepId: string): number {
  const step = plan.steps.find((s) => s.id === stepId);
  if (step === undefined) return 0;
  const users = usersOf(plan, stepId);
  const others = plan.steps.filter((s) => s.id !== stepId);
  for (let d = 1; ; d++) {
    const newStart = startOf(starts, step) + d;
    const newEnd = newStart + step.minutes;
    if (newEnd > 0) return d - 1;
    if (users.some((u) => startOf(starts, u) < newEnd)) return d - 1;
    for (let minute = newStart; minute < newEnd; minute++) {
      for (const resource of RESOURCE_NAMES) {
        if (!usesResource(step, resource)) continue;
        let count = 1;
        for (const o of others) {
          if (usesResource(o, resource) && activeAt(starts, o, minute)) count++;
        }
        if (count > limitOf(kitchen, resource)) return d - 1;
      }
    }
  }
}

export function startsOf(steps: readonly ScheduledStep[]): Starts {
  const out: Starts = {};
  for (const s of steps) out[s.id] = s.startMin;
  return out;
}

  let ovens = 0;
  for (const step of ordered) {
    let next = step;
    if (step.handsOn) {
      hands++;
      if (hands > 6) next = { ...next, handsOn: false };
    }
    if (step.appliance === 'burner') {
      burners++;
      if (burners > 8) next = { ...next, appliance: 'none' };
    }
    if (step.appliance === 'oven') {
      ovens++;
      if (ovens > 4) next = { ...next, appliance: 'none' };
    }
    changed.set(step.id, next);
  }
  return { ...plan, steps: plan.steps.map((s) => changed.get(s.id) ?? s) };
}

/**
 * Counts at least the steps needing each resource; only for plans with at most 8 burner, 4
 * oven and 6 hands-on steps (see C1). Each count is drawn from the number needed up to the
 * schema maximum (and cooks from at least 1).
 */
export function arbUnlimitedKitchen(plan: Plan): fc.Arbitrary<Kitchen> {
  const burners = plan.steps.filter((s) => s.appliance === 'burner').length;
  const ovens = plan.steps.filter((s) => s.appliance === 'oven').length;
  const cooks = plan.steps.filter((s) => s.handsOn).length;
  return fc.record({
    burners: fc.integer({ min: burners, max: 8 }),
    ovens: fc.integer({ min: ovens, max: 4 }),
    cooks: fc.integer({ min: Math.max(1, cooks), max: 6 }),
  });
}

/** `arbPlan()` fitted by `fitUnlimited`, then a kitchen from `arbUnlimitedKitchen`. */
export function arbUnlimitedPair(): fc.Arbitrary<{ plan: Plan; kitchen: Kitchen }> {
  return arbPlan()
    .map(fitUnlimited)
    .chain((plan) => arbUnlimitedKitchen(plan).map((kitchen) => ({ plan, kitchen })));
}

/** `arbPlanAndKitchen()` with every step's `handsOn` set to false. */
export function arbHandsFreePair(): fc.Arbitrary<{ plan: Plan; kitchen: Kitchen }> {
  return arbPlanAndKitchen().map(({ plan, kitchen }) => ({
    plan: { ...plan, steps: plan.steps.map((s) => ({ ...s, handsOn: false })) },
    kitchen,
  }));
}

export type Resource = 'burner' | 'oven' | 'cook';
export const RESOURCES: readonly Resource[] = ['burner', 'oven', 'cook'];
export type Capacity = Readonly<Record<Resource, number>>;

export function capacityOf(kitchen: Kitchen): Capacity {
  return { burner: kitchen.burners, oven: kitchen.ovens, cook: kitchen.cooks };
}

/** In RESOURCES order: the appliance (unless 'none'), then 'cook' iff handsOn. */
export function resourcesOf(step: Pick<Step, 'appliance' | 'handsOn'>): Resource[] {
  const out: Resource[] = [];
  if (step.appliance !== 'none') out.push(step.appliance);
  if (step.handsOn) out.push('cook');
  return out;
}

export interface Timeline {
  usage(r: Resource, minute: number): number;
  add(rs: readonly Resource[], startMin: number, endMin: number): void;
  remove(rs: readonly Resource[], startMin: number, endMin: number): void;
  /** Largest t in [startMin, endMin) with usage(r, t) + 1 > cap(r) for some r in rs; else null. */
  lastFull(rs: readonly Resource[], startMin: number, endMin: number): number | null;
  /** Smallest such t; else null. */
  firstFull(rs: readonly Resource[], startMin: number, endMin: number): number | null;
}

/**
 * Local mutable state inside one `schedule` call: one map per resource, minute to count,
 * a missing minute meaning 0. Nothing outside holds a reference to it.
 */
export function createTimeline(capacity: Capacity): Timeline {
  const counts: Record<Resource, Map<number, number>> = {
    burner: new Map(),
    oven: new Map(),
    cook: new Map(),
  };
  const usage = (r: Resource, minute: number): number => counts[r].get(minute) ?? 0;
  const isFull = (rs: readonly Resource[], minute: number): boolean =>
    rs.some((r) => usage(r, minute) + 1 > capacity[r]);
  const change = (rs: readonly Resource[], startMin: number, endMin: number, by: 1 | -1): void => {
    for (const r of rs) {
      for (let minute = startMin; minute < endMin; minute++) {
        counts[r].set(minute, usage(r, minute) + by);
      }
    }
  };
  return {
    usage,
    add: (rs, startMin, endMin) => change(rs, startMin, endMin, 1),
    remove: (rs, startMin, endMin) => change(rs, startMin, endMin, -1),
    lastFull(rs, startMin, endMin) {
      for (let minute = endMin - 1; minute >= startMin; minute--) {
        if (isFull(rs, minute)) return minute;
      }
      return null;
    },
    firstFull(rs, startMin, endMin) {
      for (let minute = startMin; minute < endMin; minute++) {
        if (isFull(rs, minute)) return minute;
      }
      return null;
    },
  };
}

/**
 * Latest end e <= upper such that [e - minutes, e) fits; throws when the start would fall
 * below floorMin (a bug: the caller's floor is below every busy minute).
 *
 * Jump: if minute t is full in the window ending at e, the windows ending at t + 1 .. e all
 * contain t when t is the last full minute of the window ending at e (they start no later than
 * the current one and end after t), so none fits; the next candidate is e = t. This gives the
 * same answer as trying every end one by one.
 */

  tl: Timeline,
  rs: readonly Resource[],

): number {

    if (end - minutes < floorMin) {
      throw new Error(`latestFit: no room at or above minute ${floorMin}`);
    }
    const full = tl.lastFull(rs, end - minutes, end);
    if (full === null) return end;
    end = full;
  }
}

/**
 * Earliest start s >= lower such that [s, s + minutes) fits; throws when the start would pass
 * ceilMin (a bug). Mirror of `latestFit`: a full minute t in the window starting at s rules out
 * every start from s to t, so the next candidate is s = t + 1.
 */
export function earliestFit(
  tl: Timeline,
  rs: readonly Resource[],
  minutes: number,
  lower: number,
  ceilMin: number,
): number {
  let start = lower;
  for (;;) {
    if (start > ceilMin) {
      throw new Error(`earliestFit: no room at or below minute ${ceilMin}`);
    }
    const full = tl.firstFull(rs, start, start + minutes);
    if (full === null) return start;
    start = full + 1;
  }
}

import { topologicalOrder } from '../graph.js';
import type { Plan, Step } from '../schema.js';
import { createTimeline, earliestFit, resourcesOf } from '../timeline.js';
import type { Capacity } from '../timeline.js';

/**
 * Pass 2. Moves every step whose result keeps (terminal or not), in topological order, to the
 * earliest start at or after both `T0` (the earliest start so far) and the latest end of its
 * prerequisites where its resources are free. Steps whose result does not keep stay put, and
 * no step moves later. `plan` must be canonical and valid.
 */
export function prepForwardPass(
  plan: Plan,
  cap: Capacity,
  starts: ReadonlyMap<string, number>,
): Map<string, number> {
  const order = topologicalOrder(plan);
  if (!order.ok) throw new Error('prepForwardPass: the plan has a cycle');
  const byId = new Map<string, Step>(plan.steps.map((s) => [s.id, s]));
  const next = new Map(starts);
  const startOf = (id: string): number => {
    const value = next.get(id);
    if (value === undefined) throw new Error(`prepForwardPass: no start for ${id}`);
    return value;
  };
  const t0 = Math.min(...next.values());
  const tl = createTimeline(cap);
  for (const step of plan.steps) {
    tl.add(resourcesOf(step), startOf(step.id), startOf(step.id) + step.minutes);
  }
  for (const id of order.value) {
    const step = byId.get(id);
    if (step === undefined || step.holdMin !== null) continue;
    const rs = resourcesOf(step);
    const current = startOf(id);
    tl.remove(rs, current, current + step.minutes);
    const lower = Math.max(
      t0,
      ...step.after.map((p) => startOf(p) + (byId.get(p)?.minutes ?? 0)),
    );
    const moved = earliestFit(tl, rs, step.minutes, lower, current);
    next.set(id, moved);
    tl.add(rs, moved, moved + step.minutes);
  }
  return new Map([...next.entries()].sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0)));
}

import { successors } from '../graph.js';
import { compareIds } from '../schema.js';
import type { Plan, Step } from '../schema.js';
import { createTimeline, latestFit, resourcesOf } from '../timeline.js';
import type { Capacity } from '../timeline.js';

/**
 * Pass 3. Moves every step whose result does not keep, latest end first (ties by id), to the
 * latest end at or before its deadline (the earliest current successor start, or 0) where its
 * resources are free. Steps whose result keeps stay put, and no step moves earlier.
 * `plan` must be canonical and valid.
 */
export function rightJustifyPass(
  plan: Plan,
  cap: Capacity,
  starts: ReadonlyMap<string, number>,
): Map<string, number> {
  const succ = successors(plan);
  const next = new Map(starts);
  const startOf = (id: string): number => {
    const value = next.get(id);
    if (value === undefined) throw new Error(`rightJustifyPass: no start for ${id}`);
    return value;
  };
  const endOf = (step: Step): number => startOf(step.id) + step.minutes;
  const tl = createTimeline(cap);
  for (const step of plan.steps) tl.add(resourcesOf(step), startOf(step.id), endOf(step));

  const order = plan.steps
    .filter((s) => s.holdMin !== null)
    .map((s) => ({ step: s, end: endOf(s) }))
    .sort((a, b) => b.end - a.end || compareIds(a.step.id, b.step.id));
  for (const { step } of order) {
    const rs = resourcesOf(step);
    const currentEnd = endOf(step);
    tl.remove(rs, startOf(step.id), currentEnd);
    const users = succ.get(step.id) ?? [];
    const deadline = users.length === 0 ? 0 : Math.min(...users.map(startOf));
    const end = latestFit(tl, rs, step.minutes, deadline, currentEnd - step.minutes);
    next.set(step.id, end - step.minutes);
    tl.add(rs, end - step.minutes, end);
  }
  return new Map([...next.entries()].sort(([a], [b]) => compareIds(a, b)));
}

import { compareIds } from '../schema.js';
import type { Plan, Step, StepLanes } from '../schema.js';
import { resourcesOf } from '../timeline.js';
import type { Capacity, Resource } from '../timeline.js';

/**
 * Pass 4. Greedy interval colouring in order of (start, id): for each resource a step uses,
 * it takes the lane of the prerequisite that ended last (ties by id) when that lane is free,
 * otherwise the smallest free lane. `plan` must be canonical and valid; `starts` must satisfy
 * capacity, so a free lane always exists (none free is a bug and throws).
 */
export function assignLanes(
  plan: Plan,
  cap: Capacity,
  starts: ReadonlyMap<string, number>,
): Map<string, StepLanes> {
  const byId = new Map<string, Step>(plan.steps.map((s) => [s.id, s]));
  const startOf = (id: string): number => {
    const value = starts.get(id);
    if (value === undefined) throw new Error(`assignLanes: no start for ${id}`);
    return value;
  };
  const endOf = (step: Step): number => startOf(step.id) + step.minutes;
  const lanes = new Map<string, StepLanes>();
  const laneOf = (id: string, r: Resource): number | null => lanes.get(id)?.[r === 'burner' ? 'burner' : r === 'oven' ? 'oven' : 'cook'] ?? null;

  const order = [...plan.steps].sort(
    (a, b) => startOf(a.id) - startOf(b.id) || compareIds(a.id, b.id),
  );
  const assigned: Step[] = [];
  for (const step of order) {
    const chosen: Record<Resource, number | null> = { burner: null, oven: null, cook: null };
    for (const r of resourcesOf(step)) {
      const busy = new Set<number>();
      for (const u of assigned) {
        if (endOf(u) <= startOf(step.id) || !resourcesOf(u).includes(r)) continue;
        const lane = laneOf(u.id, r);
        if (lane !== null) busy.add(lane);
      }
      const candidates = step.after
        .map((p) => byId.get(p))
        .filter((p): p is Step => p !== undefined && resourcesOf(p).includes(r))
        .sort((a, b) => endOf(b) - endOf(a) || compareIds(a.id, b.id));
      let lane: number | null = null;
      for (const p of candidates) {
        const l = laneOf(p.id, r);
        if (l !== null && !busy.has(l)) {
          lane = l;
          break;
        }
      }
      for (let l = 1; lane === null && l <= cap[r]; l++) {
        if (!busy.has(l)) lane = l;
      }
      if (lane === null) throw new Error(`assignLanes: no free ${r} lane for ${step.id}`);
      chosen[r] = lane;
    }
    lanes.set(step.id, chosen);
    assigned.push(step);
  }
  return new Map([...lanes.entries()].sort(([a], [b]) => compareIds(a, b)));
}

import { successors } from '../graph.js';

import type { HoldExceededIssue, Plan, Step } from '../schema.js';
import { createTimeline, resourcesOf } from '../timeline.js';
import type { Capacity } from '../timeline.js';

export interface Readout {
  slackMin: Map<string, number>;
  waitMin: Map<string, number | null>;
  warnings: HoldExceededIssue[];
}

const minutesText = (n: number): string => `${n} ${n === 1 ? 'minute' : 'minutes'}`;

export function holdExceededMessage(
  stepId: string,
  terminal: boolean,
  waitMin: number,
  holdMin: number,
): string {
  const before = terminal ? 'serving' : 'the step that uses it';
  return `Step ${stepId} will be ready ${minutesText(waitMin)} before ${before} and can wait only ${minutesText(holdMin)}. More ovens, burners or cooks may help.`;
}

/**
 * Pass 5. Slack by the G8 definition (delay one step alone, lanes and holds ignored), waits
 * measured to the latest successor, and one `hold_exceeded` per step that waits too long.
 * `plan` must be canonical and valid and `starts` must satisfy G1 to G3.
 */
export function readout(plan: Plan, cap: Capacity, starts: ReadonlyMap<string, number>): Readout {
  const succ = successors(plan);
  const byId = new Map<string, Step>(plan.steps.map((s) => [s.id, s]));
  const startOf = (id: string): number => {
    const value = starts.get(id);
    if (value === undefined) throw new Error(`readout: no start for ${id}`);
    return value;
  };
  const tl = createTimeline(cap);
  for (const s of plan.steps) tl.add(resourcesOf(s), startOf(s.id), startOf(s.id) + s.minutes);

  const slackMin = new Map<string, number>();
  const waitMin = new Map<string, number | null>();
  const warnings: HoldExceededIssue[] = [];
  for (const step of plan.steps) {
    const users = (succ.get(step.id) ?? []).map(startOf);
    const end = startOf(step.id) + step.minutes;
    const rs = resourcesOf(step);

    const limit = (users.length === 0 ? 0 : Math.min(...users)) - end;
    tl.remove(rs, startOf(step.id), end);
    let slack = limit;
    for (let d = 1; d <= limit; d++) {
      if (tl.firstFull(rs, end + d - 1, end + d) !== null) {
        slack = d - 1;
        break;
      }
    }
    tl.add(rs, startOf(step.id), end);
    slackMin.set(step.id, slack);

    const wait = step.holdMin === null ? null : (users.length === 0 ? 0 : Math.max(...users)) - end;
    waitMin.set(step.id, wait);
    if (wait !== null && step.holdMin !== null && wait > step.holdMin) {
      warnings.push({
        severity: 'warning',
        code: 'hold_exceeded',
        message: holdExceededMessage(step.id, users.length === 0, wait, step.holdMin),
        stepId: step.id,
        waitMin: wait,
        holdMin: step.holdMin,
      });
    }
  }
  void byId;
  void compareIds;
  return { slackMin, waitMin, warnings: sortIssues(warnings) as HoldExceededIssue[] };
}
