/** Equals core's `REBUILD_LIMITS.extraMax`, which `src/core/index.ts` does not export (checked in a test). */

/**
 * The instant every action records its facts at: the clock, but never before the session began
 * or before any recorded fact, so a clock set backwards cannot make a fact end before it started.
 */
export function effectiveNow(core: CookCore, nowMs: number): number {
  let t = Math.max(nowMs, core.beganAtMs);
  for (const fact of Object.values(core.facts)) {
    t = Math.max(t, fact.startedAtMs);
    if (fact.state === 'done') t = Math.max(t, fact.endedAtMs);
  }
  return t;
}

/** The facts as core `Progress` on the session's axis. */
export function progressOf(core: CookCore): Progress {
  const progress: Progress = {};
  for (const [id, fact] of Object.entries(core.facts)) {
    const startMin = minuteAt(core.axisMs, fact.startedAtMs);
    const entry: ProgressEntry =
      fact.state === 'done'
        ? { state: 'done', startMin, endMin: minuteAt(core.axisMs, fact.endedAtMs) }
        : { state: 'running', startMin, extraMin: fact.extraMin };
    progress[id] = entry;
  }
  return progress;
}

/** When a running step is ready: its recorded start plus its minutes plus the extra minutes. */
export function endsAtMs(step: Step, fact: Extract<StepFact, { state: 'running' }>): number {
  return clockAt(fact.startedAtMs, step.minutes + fact.extraMin);
}

/**
 * Rebuilds `core` against its facts at axis minute `n` and appends the log entry. A failed
 * rebuild keeps the schedule and the facts and logs the error kind (CM5.7).

    schedule: result.ok ? result.value.schedule : core.schedule,

/** The session a cook would begin now, or why not. `too_late`: the rebuild is out of range. */
export function beginSession(
  schedule: Schedule | RebuiltSchedule,
  axisMs: number,
  nowMs: number,
): Result<CookState, BeginError> {
  const parsed = RebuiltScheduleSchema.safeParse(schedule);
  if (!parsed.success) return err({ kind: 'invalid_schedule' });
  const planned = parsed.data;
  let core: CookCore = {
    axisMs,
    plannedServingMs: clockAt(axisMs, planned.servingMin),
    beganAtMs: nowMs,
    schedule: planned,
    facts: {},
    log: [],
  };
  const n = minuteAt(axisMs, nowMs);
  const startMin = earliestStartMin(planned);
  if (n > startMin) {
    const result = rebuild(planned, {}, n);
    if (!result.ok) {
      const kind = result.error.kind;
      return err({
        kind: kind === 'invalid_now' || kind === 'serving_out_of_range' ? 'too_late' : 'invalid_schedule',
      });

    core = {
      ...core,
      schedule: result.value.schedule,
      log: [entryFor(nowMs, { kind: 'begin-late', byMin: n - startMin }, planned, result.value)],

  | { do: 'extra'; id: string; minutes: 5 | 10 | 15 }
  | { do: 'behind' }
);

/** Applies the steps in order with `applyCook`. */
function run(state: CookState, ...steps: Step[]): CookState {
  let current = state;
  for (const step of steps) {
    if (step.do === 'behind') {
      current = applyCook(current, { type: 'cook/behind', nowMs: step.at });
    } else if (step.do === 'extra') {
      current = applyCook(current, {
        type: 'cook/extra',
        stepId: step.id,
        minutes: step.minutes,
        nowMs: step.at,
      });
    } else {
      current = applyCook(current, {
        type: step.do === 'start' ? 'cook/start' : 'cook/done',
        stepId: step.id,
        nowMs: step.at,
      });
    }
  }
  return current;
}

function lastEntry(state: CookState): LogEntry {
  const entry = state.core.log.at(-1);
  expect(entry).not.toBeUndefined();
  if (entry === undefined) throw new Error('no log entry');
  return entry;
}

// ---------------------------------------------------------------------------------------
// Cooking-mode oracles (cooking-mode spec). Plain loops over steps; the minute of an instant is
// written out here, not taken from src/web/state/clock.ts.

/** What the cooking oracles need of a schedule: planned starts, ends and slack per step. */
export interface CookOracleSchedule {
  steps: readonly { id: string; startMin: number; endMin: number; slackMin: number }[];
}

/** What the cooking oracles need of a fact: only whether the step is running or done. */
export type CookOracleFacts = Readonly<Record<string, { state: 'running' | 'done' }>>;

const MS_PER_MINUTE = 60_000;

/**
 * CM2.2 to CM2.6: the ids of the steps that are due (pending, planned start at or before now),
 * running, and the first three pending steps that are not due by (planned start, id).
 */
export function oracleCookSections(
  schedule: CookOracleSchedule,
  facts: CookOracleFacts,
  axisMs: number,
  nowMs: number,
): { due: string[]; running: string[]; upNext: string[] } {
  const due: string[] = [];
  const running: string[] = [];
  const later: { id: string; startMs: number }[] = [];
  for (const row of schedule.steps) {
    const fact = facts[row.id];
    if (fact?.state === 'done') continue;
    if (fact?.state === 'running') {
      running.push(row.id);
      continue;
    }
    const startMs = axisMs + row.startMin * MS_PER_MINUTE;
    if (startMs <= nowMs) due.push(row.id);
    else later.push({ id: row.id, startMs });
  }
  later.sort((a, b) => a.startMs - b.startMs || (a.id < b.id ? -1 : a.id > b.id ? 1 : 0));
  return {
    due: due.sort(),
    running: running.sort(),
    upNext: later.slice(0, 3).map((x) => x.id),
  };
}

/**
 * The late steps, sorted by id. With `n` the floor of the minutes since the axis instant: a
 * running step is late when `n >= endMin + 5`; a pending one when `n >= startMin + max(5, slack)`.
 */
export function oracleOverdue(
  schedule: CookOracleSchedule,
  facts: CookOracleFacts,
  axisMs: number,
  nowMs: number,
): string[] {
  const n = Math.floor((nowMs - axisMs) / MS_PER_MINUTE);
  const late: string[] = [];
  for (const row of schedule.steps) {
    const fact = facts[row.id];
    if (fact?.state === 'done') continue;
    if (fact?.state === 'running') {
      if (n >= row.endMin + 5) late.push(row.id);
    } else if (n >= row.startMin + Math.max(5, row.slackMin)) {
      late.push(row.id);
    }
  }
  return late.sort();
}

/**
 * Ids of the hands-free steps that are running and whose time came up in `(fromMs, toMs]`,
 * by end time then id (CM4.2).
 */
export function finishedBetween(state: CookState, fromMs: number, toMs: number): string[] {
  const { core } = state;
  const ended: { id: string; at: number }[] = [];
  for (const step of core.schedule.plan.steps) {
    const fact = core.facts[step.id];
    if (step.handsOn || fact?.state !== 'running') continue;
    const at = endsAtMs(step, fact);
    if (at > fromMs && at <= toMs) ended.push({ id: step.id, at });
  }
  return ended.sort((a, b) => a.at - b.at || compareIds(a.id, b.id)).map((e) => e.id);
}

## Decisions I made

- **"Start now" rounds up.** The design said `round`. With `round`, a tap at 17:30:40 would still leave 20 s of countdown. `ceil` makes "Start now" start now and moves the serving time at most 59 s further than the exact gap. The `design.md` table row is updated.
- **The expected new warning after a late begin is only `greens.toss`.** T1.3 also named the roast, but the roast ends when the rest starts, so it waits 0 minutes. That matches the design's hand-worked CM-b and my test. I corrected the T1.3 text.
- **`EXTRA_MAX = 1440` is a local constant, with a test that it equals core's `REBUILD_LIMITS.extraMax`.** Core's public surface is pinned by a test and doesn't export that constant. I left core alone.
- **P-CM7 also allows `invalid_now` as a failed-rebuild kind.** It is the same "past the 24-hour window" case as `serving_out_of_range`, and the property asserts the minute really is above 1440.
- **Reducers ignore bad input.** `applyCook` and `beginSession` return the state unchanged for a non-integer or negative instant, a non-boolean `wanted`, or an unknown action type. Otherwise a wild value could corrupt a stored session.
- **`announcementFor` takes an optional `locale`**, like the other layout functions, so tests can use `en-GB`.
- **`finishedBetween` returns step ids**, not titles.
